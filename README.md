NovaStock — Aplicación web CRUD con C# y Oracle Database 19c:
Aplicación web de ejemplo desarrollada para Visual Studio 2026, con ASP.NET Core MVC, C#, Bootstrap 5 y Oracle Database 19c. 
Permite crear, consultar, editar y eliminar productos mediante una interfaz web moderna .
Nota sobre versiones: el proyecto está configurado para net10.0. Si tu instalación de Visual Studio usa otro SDK, cambia 
TargetFramework en el .csproj por el framework instalado y compatible. 
La conexión usa el paquete Oracle.ManagedDataAccess.Core.

1. Funcionalidades
- Listado de productos almacenados en Oracle.
- Búsqueda por nombre o categoría.
- Creación, edición y eliminación de productos.
- Validaciones de datos en el servidor.
- Tarjetas con total de productos, unidades en stock y valor del inventario.
- Confirmación antes de eliminar.
- Interfaz adaptable a escritorio, tablet y móvil.
- Consultas parametrizadas para reducir el riesgo de inyección SQL.

5.5 Data/ProductoRepository.cs
using CrudProductosOracle19c.Models;
using Oracle.ManagedDataAccess.Client;

namespace CrudProductosOracle19c.Data;

public class ProductoRepository
{
    private readonly string _connectionString;

    public ProductoRepository(IConfiguration configuration)
    {
        _connectionString = configuration.GetConnectionString("OracleDb")
            ?? throw new InvalidOperationException(
                "Falta ConnectionStrings:OracleDb en appsettings.json.");
    }

    private OracleConnection CrearConexion() => new(_connectionString);

    public async Task<List<Producto>> ObtenerTodosAsync(string? buscar = null)
    {
        var productos = new List<Producto>();
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        const string sql = """
            SELECT ID, NOMBRE, DESCRIPCION, PRECIO, STOCK, CATEGORIA, FECHA_CREACION
            FROM PRODUCTOS
            WHERE (:p_buscar IS NULL
               OR UPPER(NOMBRE) LIKE '%' || UPPER(:p_buscar) || '%'
               OR UPPER(CATEGORIA) LIKE '%' || UPPER(:p_buscar) || '%')
            ORDER BY ID DESC
            """;

        await using var cmd = new OracleCommand(sql, cn);
        cmd.BindByName = true;
        cmd.Parameters.Add("p_buscar", OracleDbType.Varchar2).Value =
            string.IsNullOrWhiteSpace(buscar) ? DBNull.Value : buscar.Trim();

        await using var reader = await cmd.ExecuteReaderAsync();
        while (await reader.ReadAsync())
            productos.Add(LeerProducto(reader));

        return productos;
    }

    public async Task<Producto?> ObtenerPorIdAsync(int id)
    {
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        const string sql = """
            SELECT ID, NOMBRE, DESCRIPCION, PRECIO, STOCK, CATEGORIA, FECHA_CREACION
            FROM PRODUCTOS
            WHERE ID = :p_id
            """;

        await using var cmd = new OracleCommand(sql, cn);
        cmd.BindByName = true;
        cmd.Parameters.Add("p_id", OracleDbType.Int32).Value = id;

        await using var reader = await cmd.ExecuteReaderAsync();
        return await reader.ReadAsync() ? LeerProducto(reader) : null;
    }

    public async Task CrearAsync(Producto producto)
    {
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        const string sql = """
            INSERT INTO PRODUCTOS
                (NOMBRE, DESCRIPCION, PRECIO, STOCK, CATEGORIA)
            VALUES
                (:p_nombre, :p_descripcion, :p_precio, :p_stock, :p_categoria)
            """;

        await using var cmd = new OracleCommand(sql, cn);
        cmd.BindByName = true;
        AgregarParametrosProducto(cmd, producto);
        await cmd.ExecuteNonQueryAsync();
    }

    public async Task<bool> ActualizarAsync(Producto producto)
    {
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        const string sql = """
            UPDATE PRODUCTOS
            SET NOMBRE = :p_nombre,
                DESCRIPCION = :p_descripcion,
                PRECIO = :p_precio,
                STOCK = :p_stock,
                CATEGORIA = :p_categoria
            WHERE ID = :p_id
            """;

        await using var cmd = new OracleCommand(sql, cn);
        cmd.BindByName = true;
        AgregarParametrosProducto(cmd, producto);
        cmd.Parameters.Add("p_id", OracleDbType.Int32).Value = producto.Id;

        return await cmd.ExecuteNonQueryAsync() > 0;
    }

    public async Task<bool> EliminarAsync(int id)
    {
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        await using var cmd =
            new OracleCommand("DELETE FROM PRODUCTOS WHERE ID = :p_id", cn);
        cmd.BindByName = true;
        cmd.Parameters.Add("p_id", OracleDbType.Int32).Value = id;

        return await cmd.ExecuteNonQueryAsync() > 0;
    }

    public async Task<int> ContarAsync()
    {
        await using var cn = CrearConexion();
        await cn.OpenAsync();

        await using var cmd = new OracleCommand(
            "SELECT COUNT(*) FROM PRODUCTOS", cn);

        return Convert.ToInt32(await cmd.ExecuteScalarAsync());
    }

    private static void AgregarParametrosProducto(
        OracleCommand cmd, Producto p)
    {
        cmd.Parameters.Add("p_nombre", OracleDbType.Varchar2, 100).Value =
            p.Nombre.Trim();
        cmd.Parameters.Add("p_descripcion", OracleDbType.Varchar2, 500).Value =
            p.Descripcion.Trim();
        cmd.Parameters.Add("p_precio", OracleDbType.Decimal).Value = p.Precio;
        cmd.Parameters.Add("p_stock", OracleDbType.Int32).Value = p.Stock;
        cmd.Parameters.Add("p_categoria", OracleDbType.Varchar2, 30).Value =
            p.Categoria.Trim();
    }

    private static Producto LeerProducto(OracleDataReader r) => new()
    {
        Id = Convert.ToInt32(r["ID"]),
        Nombre = r["NOMBRE"].ToString() ?? "",
        Descripcion = r["DESCRIPCION"].ToString() ?? "",
        Precio = Convert.ToDecimal(r["PRECIO"]),
        Stock = Convert.ToInt32(r["STOCK"]),
        Categoria = r["CATEGORIA"].ToString() ?? "",
        FechaCreacion = Convert.ToDateTime(r["FECHA_CREACION"])
    };
}

5.6 Controllers/ProductosController.cs
using CrudProductosOracle19c.Data;
using CrudProductosOracle19c.Models;
using Microsoft.AspNetCore.Mvc;

namespace CrudProductosOracle19c.Controllers;

public class ProductosController : Controller
{
    private readonly ProductoRepository _repository;

    public ProductosController(ProductoRepository repository)
        => _repository = repository;

    public async Task<IActionResult> Index(string? buscar)
    {
        ViewBag.Buscar = buscar;
        ViewBag.TotalProductos = await _repository.ContarAsync();
        return View(await _repository.ObtenerTodosAsync(buscar));
    }

    public IActionResult Crear() => View(new Producto());

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Crear(Producto producto)
    {
        if (!ModelState.IsValid) return View(producto);

        await _repository.CrearAsync(producto);
        TempData["Mensaje"] = "El producto se creó correctamente.";
        return RedirectToAction(nameof(Index));
    }

    public async Task<IActionResult> Editar(int id)
    {
        var producto = await _repository.ObtenerPorIdAsync(id);
        return producto is null ? NotFound() : View(producto);
    }

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Editar(Producto producto)
    {
        if (!ModelState.IsValid) return View(producto);

        var actualizado = await _repository.ActualizarAsync(producto);
        if (!actualizado) return NotFound();

        TempData["Mensaje"] = "El producto se actualizó correctamente.";
        return RedirectToAction(nameof(Index));
    }

    public async Task<IActionResult> Eliminar(int id)
    {
        var producto = await _repository.ObtenerPorIdAsync(id);
        return producto is null ? NotFound() : View(producto);
    }

    [HttpPost, ActionName("Eliminar")]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> ConfirmarEliminar(int id)
    {
        var eliminado = await _repository.EliminarAsync(id);
        TempData["Mensaje"] = eliminado
            ? "El producto se eliminó correctamente."
            : "No se encontró el producto solicitado.";
        return RedirectToAction(nameof(Index));
    }
}

6. Crear y ejecutar el proyecto en Visual Studio
  1. Abre Visual Studio 2026.
  2. Selecciona Crear un proyecto.
  3. Elige ASP.NET Core Web App (Model-View-Controller).
  4. Usa el nombre CrudProductosOracle19c.
  5. Selecciona el framework .NET instalado que vayas a utilizar.
  6. Crea las carpetas y archivos siguiendo la estructura anterior. Reemplaza el contenido de los archivos indicados.
  7. En Herramientas → Administrador de paquetes NuGet, instala Oracle.ManagedDataAccess.Core si el paquete no se restaura automáticamente.
  8. Ejecuta los scripts SQL de la sección 4 en Oracle SQL Developer.
  9. Actualiza appsettings.json con el usuario, contraseña y Data Source correctos.
  10. En Visual Studio, selecciona Compilar → Compilar solución.
  11. Ejecuta con F5 o Ctrl+F5. La ruta inicial debe abrir la página de Productos.

7. Configurar la conexión Oracle
Ejemplo de cadena de conexión:
User Id=CRUD_APP;Password=TU_PASSWORD;Data Source=localhost:1521/XEPDB1;
- localhost: servidor de Oracle.
- 1521: puerto de escucha habitual, si no lo has cambiado.
- XEPDB1: ejemplo de service name; puede ser distinto en tu instalación.
- CRUD_APP: usuario de base de datos de ejemplo.
- TU_PASSWORD: contraseña configurada para ese usuario.
Si tu instalación usa otro service name, consulta la configuración de Oracle y reemplaza XEPDB1.
Verifica que el listener esté iniciado y que el usuario tenga permiso para consultar e insertar en PRODUCTOS.

8. Pruebas del CRUD
  1. Consultar: abre la página inicial y verifica que aparezcan los productos de prueba.
  2. Buscar: escribe teclado o Tecnología en el buscador.
  3. Crear: pulsa Nuevo producto, completa el formulario y guarda.
  4. Editar: pulsa el icono de edición, cambia precio o stock y guarda.
  5. Eliminar: pulsa el icono de eliminación y confirma.
  6. Verificar en Oracle: ejecuta:

SELECT ID, NOMBRE, PRECIO, STOCK, CATEGORIA
FROM PRODUCTOS
ORDER BY ID DESC;

Los cambios realizados desde la web deben verse en Oracle.
9. Solución de problemas comunes
- ORA-12154 / ORA-12514: revisa el Data Source, el service name y el estado del listener.
- ORA-01017: el usuario o la contraseña no son correctos.
- ORA-00942: la tabla PRODUCTOS no existe en el esquema conectado o estás usando otro usuario.
- Error de paquete NuGet: restaura los paquetes y selecciona una versión compatible de Oracle.ManagedDataAccess.Core.
- No se abre la página: revisa la salida de Visual Studio, el perfil de lanzamiento y el puerto asignado.
- La interfaz no carga estilos: se requiere conexión a internet para las hojas de estilo CDN de Bootstrap y Google Fonts; el CSS propio se encuentra en wwwroot/css/site.css.

10. Recomendaciones para un entorno real
- No publiques contraseñas en appsettings.json ni en Git. Usa Secret Manager o variables de entorno.
- Configura autenticación y autorización antes de exponer el sistema.
- Habilita HTTPS y un manejo de errores apropiado para producción.
- Agrega registros de auditoría y copias de seguridad según los requisitos del negocio.
- Revisa la versión del proveedor Oracle aprobada para tu entorno.
